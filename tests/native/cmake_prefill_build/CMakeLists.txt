# Configure the real build graph without a CANN installation or C++ compiler.
cmake_minimum_required(VERSION 3.16)
project(prefill_build_regression NONE)
if(NOT PREFILL_REPO_ROOT)
  message(FATAL_ERROR "Pass -DPREFILL_REPO_ROOT=/path/to/LMCache-Ascend")
endif()
set(PREFILL_FIXTURE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_SYSTEM_PROCESSOR aarch64)

# Only SDK/package/compiler boundaries are mocked. Both production CMakeLists
# and the vendor's source enumeration and npu_lib.cmake execute unchanged.
macro(project)
  set(PROJECT_SOURCE_DIR "${PREFILL_REPO_ROOT}")
endmacro()
macro(find_package)
endmacro()
macro(install)
endmacro()
macro(target_link_options)
endmacro()
macro(target_link_directories)
endmacro()
function(target_link_libraries target)
  set_property(TARGET ${target} APPEND PROPERTY PREFILL_LINKS ${ARGN})
endfunction()
function(target_include_directories target visibility)
  _target_include_directories(${target} INTERFACE ${ARGN})
endfunction()

# A macro intentionally preserves include()'s caller-scope semantics.
macro(include path)
  if("${path}" MATCHES "(^|/)ascendc_with_def\\.cmake$")
    _include("${PREFILL_FIXTURE_DIR}/fake_ascendc.cmake")
  else()
    _include("${path}" ${ARGN})
  endif()
endmacro()
function(add_subdirectory directory)
  if(directory STREQUAL "csrc" OR directory STREQUAL "csrc/mindspore")
    add_library(c_ops INTERFACE)
  else()
    _add_subdirectory("${PREFILL_REPO_ROOT}/${directory}"
                     "${CMAKE_BINARY_DIR}/${directory}")
  endif()
endfunction()

include("${PREFILL_REPO_ROOT}/CMakeLists.txt")

if(NOT TARGET host_intf_pub OR NOT TARGET cache_kernels OR NOT TARGET c_ops)
  message(FATAL_ERROR "Missing SDK or production library target")
endif()
if(TARGET prefill_load_kernels)
  message(FATAL_ERROR "Prefill kernels must use the existing shared library")
endif()
get_target_property(kernel_type cache_kernels PREFILL_TYPE)
if(NOT kernel_type STREQUAL "SHARED")
  message(FATAL_ERROR "Existing cache_kernels ABI must remain shared")
endif()
get_target_property(actual cache_kernels PREFILL_SOURCES)
# Execute the vendor's own file(GLOB) in its scope and compare source coverage.
get_property(vendor_sources GLOBAL PROPERTY PREFILL_VENDOR_SOURCES)
if(DEFINED ENV{USE_MINDSPORE})
  set(expected ${vendor_sources})
else()
  # Evaluate the vendor CMake source enumeration without creating a second SDK.
  set(PREFILL_ENUMERATE_ONLY TRUE)
  _add_subdirectory("${PREFILL_REPO_ROOT}/third_party/kvcache-ops"
                   "${CMAKE_BINARY_DIR}/enumerate_vendor")
  get_property(vendor_sources GLOBAL PROPERTY PREFILL_VENDOR_SOURCES)
  set(expected ${vendor_sources}
      "${PREFILL_REPO_ROOT}/csrc/prefill_load/prefill_load.cpp")
endif()
list(SORT expected)
list(SORT actual)
if(NOT "${expected}" STREQUAL "${actual}")
  message(FATAL_ERROR "Kernel source coverage changed: expected=${expected}; actual=${actual}")
endif()
get_target_property(links c_ops PREFILL_LINKS)
if(NOT "cache_kernels" IN_LIST links OR "prefill_load_kernels" IN_LIST links)
  message(FATAL_ERROR "c_ops must link the single cache_kernels library")
endif()
if(DEFINED ASCEND_AICORE_ARCH)
  get_target_property(defs cache_kernels PREFILL_DEFINITIONS)
  if(NOT "-DASCEND_AICORE_ARCH=${ASCEND_AICORE_ARCH}" IN_LIST defs)
    message(FATAL_ERROR "Lost device/host architecture definition")
  endif()
endif()
message(STATUS "PREFILL_BUILD_GRAPH_PASSED")
